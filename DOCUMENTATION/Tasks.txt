create or replace table trgt_tbl_1(
  id int,
  name varchar,
  created_date date,
  created_day varchar,
  created_month varchar,
  created_year varchar
);

select * from trgt_tbl_1;

insert into trgt_tbl_1
select 
a.id,
a.name,
a.created_date,
day(a.created_date) as created_day,
month(a.created_date) as created_month,
year(a.created_date) as created_year
from src_tbl_1 as a
left join trgt_tbl_1 as b
on a.id = b.id;

warehouse = compute_wh,
schedule = '1 minute'
as
insert into trgt_tbl_2
select
a.id,
a.name,
a.created_date,
day(a.created_date) as created_day,
month(a.created_date) as created_month,
year(a.created_date) as created_month
from src_tbl_2 as a

on a.id = b.id ;

select * from table(information_schema.task_history(task_name => 'copy_dat'));
