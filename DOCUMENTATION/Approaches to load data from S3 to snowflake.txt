Approach - 1
------------- 
Loading by using access id and access key from S3
----------------------------------------------------

create database sfp_db;
create schema raw;
use database sfp_db;
use schema raw;

create or replace stage s3_stage
url = 's3://test-project-18052026/orders/'
credentials = (AWS_KEY_ID = 'AKIATOH23E5AAXW65566'
               AWS_SECRET_KEY = '6AzJJTLiMWB40IFBkTlvPbIfVphvR3hLuBZfYUIO');

LS @S3_STAGE;

CREATE OR REPLACE TABLE ORDERS_RAW(
 ORDER_ID NUMBER,
 CUSTOMER_ID VARCHAR(10),
 CUSTOMER_NAME VARCHAR(100),
 PRODUCT_NAME VARCHAR(100),
 CATEGORY VARCHAR(100),
 QUANTITY NUMBER,
 UNIT_PRICE FLOAT,
 TOTAL_AMOUNT FLOAT,
 ORDER_DATE DATE,
 STATUS VARCHAR(20)
);

COPY INTO ORDERS_RAW
FROM @S3_STAGE
FILE_FORMAT = (
    TYPE = 'CSV'
    FIELD_OPTIONALLY_ENCLOSED_BY = '"'
    SKIP_HEADER = 1
)
ON_ERROR = 'CONTINUE';

SELECT * FROM ORDERS_RAW;

SELECT 
CATEGORY, 
COUNT(*) AS TOTAL_ORDERS,
SUM(TOTAL_AMOUNT) AS REVENUE
FROM ORDERS_RAW
GROUP BY CATEGORY;

** IF U WANT TO LOAD THE SAME DATA RELATED TO THE SAME FIELD THERE IS NO NEED OF CREATING A NEW TABLE IF THE DATA IS NOT THE SAME U HAVE TO CREATE NEW ONE 

CREATE OR REPLACE PIPE <ANY_PIPE_NAME>
AUTO_INGEST = TRUE AS
COPY INTO ORDERS_RAW
FROM @AMAZ_STG
FILE_FORMAT = (
    TYPE = 'CSV'
    FIELD_OPTIONALLY_ENCLOSED_BY = '"'
    SKIP_HEADER = 1
)
ON_ERROR = 'CONTINUE';


SELECT SYSTEM$PIPE_STATUS(PIPE_NAME_U_GAVE);

SHOW PIPES;
