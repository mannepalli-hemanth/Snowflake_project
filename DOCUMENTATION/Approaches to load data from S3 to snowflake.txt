Approach - 1
------------- 
Loading by using access id and access key from S3
----------------------------------------------------

CREATE DATABASE MIN_PRJT_DB;
CREATE SCHEMA RAW;

DROP STAGE IF EXISTS AWS_STG;
DROP STORAGE INTEGRATION IF EXISTS AWS_ING;



DESC STORAGE INTEGRATION AWS_ING;

CREATE OR REPLACE STAGE AWS_STG
URL = 's3://test-project-18052026/orders/'
CREDENTIALS = (AWS_KEY_ID = 'XXXXXXXXXXXXXXXX' 
               AWS_SECRET_KEY ='XXXXXXXXXXXXXXXXXXXXXXXXXX');

LS @AWS_STG;

CREATE OR REPLACE TABLE ORDERS_RAW(
 ORDER_ID NUMBER,
 CUSTOMER_ID VARCHAR(10),
 CUSTOMER_NAME VARCHAR(100),
 PRODUCT_NAME VARCHAR(100),
 CATEGORY VARCHAR(100),
 QUANTITY NUMBER,
 UNIT_PRICE FLOAT,
 TOTAL_AMOUNT FLOAT,
 ORDER_DATE DATE,
 STATUS VARCHAR(20)
);

SELECT CURRENT_DATABASE(),
       CURRENT_SCHEMA(),
       CURRENT_WAREHOUSE();

USE DATABASE MIN_PRJT_DB;
USE SCHEMA RAW;

COPY INTO ORDERS_RAW
FROM @AWS_STG
FILE_FORMAT = (
    TYPE = 'CSV'
    FIELD_OPTIONALLY_ENCLOSED_BY = '"'
    SKIP_HEADER = 1
)
ON_ERROR = 'CONTINUE';

SELECT * FROM ORDERS_RAW;

SELECT 
CATEGORY, 
COUNT(*) AS TOTAL_ORDERS,
SUM(TOTAL_AMOUNT) AS REVENUE
FROM ORDERS_RAW
GROUP BY CATEGORY;

Continuous loading using Snowpipe 
---------------------------------

USE DATABASE min_prjt_db;;
USE SCHEMA raw;


CREATE STORAGE INTEGRATION (STG_ANY NAME)
TYPE = EXTERNAL_STAGE
ENABLED = TRUE
STORAGE_PROVIDER = S3
STORAGE_ALLOWED_LOCATIONS = ('XXXXXXXXXXXXXXXXXXXXXXX')
STORAGE_AWS_ROLE_ARN = 'XXXXXXXXXXXXXXXXXXXXXXXXX';

DESC STORAGE INTEGRATION AMAZ_ING;

CREATE OR REPLACE STAGE ANY_STAGE_NAME
URL = 's3://test-project-18052026/orders/'
STORAGE_INTEGRATION = THE NAME OF STG_INTEGRATION U GIVE;

LS @STAGE_NAME_GAVE;

** IF U WANT TO LOAD THE SAME DATA RELATED TO THE SAME FIELD THERE IS NO NEED OF CREATING A NEW TABLE IF THE DATA IS NOT THE SAME U HAVE TO CREATE NEW ONE 

CREATE OR REPLACE PIPE <ANY_PIPE_NAME>
COPY INTO ORDERS_RAW
FROM @AMAZ_STG
FILE_FORMAT = (
    TYPE = 'CSV'
    FIELD_OPTIONALLY_ENCLOSED_BY = '"'
    SKIP_HEADER = 1
)
ON_ERROR = 'CONTINUE';


SELECT SYSTEM$PIPE_STATUS(PIPE_NAME_U_GAVE);

SHOW PIPES;
